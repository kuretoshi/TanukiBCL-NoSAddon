using TanukiBCL.Roles;

void Check(bool value,string name) { if (!value) throw new Exception(name); }
bool Near(double a,double b) => Math.Abs(a-b)<.000001;
var radar = new SonarmanRound();
Check(radar.Activate(1,100,10,30,2),"Activate duration and sweep");
Check(!radar.Activate(2,101,10,30,2),"Cannot overlap activations");
Check(Near(radar.Remaining(101),9) && Near(radar.Phase(101),.5),"Duration and rotating sweep phase");
radar.Observe(new(1,2,0,100.5),1,0,0,7,100.5);
radar.Observe(new(2,4,0,99.5),1,0,0,7,100.5);
radar.Observe(new(3,8,0,100.5),1,0,0,7,100.5);
radar.Observe(new(4,double.NaN,0,100.5),1,0,0,7,100.5);
Check(radar.Marks.Count==0 && !radar.Sweep(101.99,0,0,7),"Self, stale, outside-range and invalid samples never update before a sweep");
radar.Observe(new(2,4,0,100.5),1,0,0,7,100.5);
radar.Observe(new(3,0,3,101.2),1,0,0,7,101.2);
Check(radar.Sweep(102,0,0,7) && radar.Marks.Count==2,"One full revolution publishes all detected directions together");
Check(radar.Marks.Any(p=>Near(p.X,1)&&Near(p.Y,0)) && radar.Marks.Any(p=>Near(p.X,0)&&Near(p.Y,1)),"Directions project to the circumference without distance information");
radar.Observe(new(2,0,-4,102.5),1,0,0,7,102.5);
Check(radar.Marks.Count==2 && !radar.Sweep(103.99,0,0,7),"Positions remain fixed until the following sweep");
Check(radar.Sweep(104,0,0,7) && radar.Marks.Count==1 && Near(radar.Marks[0].Y,-1),"New sweep updates a speaker's position and removes silent speakers");
Check(radar.Sweep(106,0,0,7) && radar.Marks.Count==0,"Empty sweep clears all stale dots");
radar.Observe(new(2,3,0,106.5),1,0,0,7,106.5);
Check(radar.Sweep(108,1,1,7) && Near(radar.Marks[0].X,2/Math.Sqrt(5)) && Near(radar.Marks[0].Y,-1/Math.Sqrt(5)),"Direction uses the recorded speech position relative to the listener at refresh");
Check(!radar.Active(110) && radar.Stop(110) && radar.Marks.Count==0 && Near(radar.ReadyAt,140),"Expiry removes markers and starts the configured cooldown");
Check(!radar.Activate(2,139,10,30,2) && !radar.Activate(1,140,10,30,2),"Cooldown and activation replay rejection");
Check(radar.Activate(2,140,10,30,2),"Ready exactly at cooldown boundary");
radar.Observe(new(2,2,0,139),1,0,0,7,140);
Check(radar.Sweep(142,0,0,7) && radar.Marks.Count==0,"A previous activation's signals cannot leak into a new scan");
radar.Observe(new(2,2,0,142.2),1,0,0,7,142.2);
Check(radar.Stop(143) && Near(radar.ReadyAt,173) && radar.Marks.Count==0,"Meeting/death cancellation clears data and starts cooldown at interruption");
Check(!radar.Stop(150) && Near(radar.ReadyAt,173),"Repeated cleanup does not restart cooldown");
radar.Clear();
Check(!radar.Running && radar.Marks.Count==0,"Release clears the radar");
var duplicate = new SonarmanRound(); duplicate.Activate(1,0,20,0,2);
duplicate.Observe(new(2,4,0,.5),1,0,0,7,.5);
duplicate.Observe(new(2,0,4,1.5),1,0,0,7,1.5);
duplicate.Observe(new(2,-4,0,1),1,0,0,7,1.5);
Check(duplicate.Sweep(2,0,0,7) && duplicate.Marks.Count==1 && Near(duplicate.Marks[0].Y,1),"Repeated speech stores only the latest position and rejects out-of-order pulses");
duplicate.Observe(new(2,2,0,2.5),1,0,0,7,2.5);
Check(duplicate.Sweep(8,0,0,7) && duplicate.Marks.Count==0,"Missed frames do not resurrect speech from old revolutions");
var moved = new SonarmanRound(); moved.Activate(1,0,10,0,2);
moved.Observe(new(2,3,0,1),1,0,0,7,1);
Check(moved.Sweep(2,20,20,7) && moved.Marks.Count==0,"Moving out of range before refresh drops the mark");
var shortScan = new SonarmanRound();shortScan.Activate(1,0,1,0,2);
shortScan.Observe(new(2,3,0,.5),1,0,0,7,.5);
Check(!shortScan.Sweep(1,0,0,7) && shortScan.Marks.Count==0,"An effect ending before a full revolution produces no premature result");
foreach(var duration in new[]{double.NaN,double.PositiveInfinity,0d,121d})
    Check(!new SonarmanRound().Activate(1,0,duration,30,2),"Invalid durations are rejected");
Check(!new SonarmanRound().Activate(1,0,10,30,0) && !new SonarmanRound().Activate(1,0,10,-1,2),"Invalid period and cooldown are rejected");
var ping = SonarmanPingSynth.Create();
Check(ping.Length==SonarmanPingSynth.Rate*SonarmanPingSynth.Seconds && SonarmanPingSynth.Seconds<.5,"Ping finishes before the minimum sweep period");
Check(ping.All(float.IsFinite) && ping.Max(x=>Math.Abs(x)) is >.1f and <1f,"Audible finite ping never clips");
Check(ping[0]==0 && ping[^1]==0 && ping.TakeLast(240).Max(x=>Math.Abs(x))<.01f,"Ping starts and ends at zero with a quiet tail");
Check(ping.SequenceEqual(SonarmanPingSynth.Create()),"Synthesized sound is deterministic");
Console.WriteLine("PASS Sonarman: radar lifecycle, refresh, directions, range, stale signal rejection; ping duration, finite unclipped samples and fade");
