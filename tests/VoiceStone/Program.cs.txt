using TanukiBCL.Roles;

void Check(bool ok, string name) { if (!ok) throw new Exception(name); }
Check(VoiceStoneRules.TryWord("ねこ", 8, out var word) && word == "ねこ", "Japanese word");
Check(VoiceStoneRules.Width("インポスター", .5f) == 3f && VoiceStoneRules.Width("ねこ", .5f) == 1f, "Character-proportional width");
Check(VoiceStoneRules.TryWord("ＴＡＳＫ", 4, out word) && word == "TASK", "Compatibility normalization");
foreach (var invalid in new[] { "", "猫 犬", "<b>猫", "猫\n", "🐈", "長すぎる単語です", "\ud800" })
    Check(!VoiceStoneRules.TryWord(invalid, 4, out _), "Reject invalid or oversized word: " + invalid);
var state = new VoiceStoneRequest();
Check(state.Arm(1, 100, 5), "Begin listening");
Check(!state.Arm(2, 101, 5), "Cannot replace an active request");
Check(!state.Accept(2, 102, "猫", 8, out _), "Reject another request");
Check(!state.Accept(1, 102, "猫 犬", 8, out _) && state.Waiting, "Bad word does not consume request");
Check(state.Accept(1, 102, "猫", 8, out word) && word == "猫" && !state.Waiting, "Recognized word places immediately before the timer ends");
Check(!state.Accept(1, 103, "猫", 8, out _), "Replay cannot create a second stone");
Check(!state.Arm(1, 120, 5) && state.Arm(2, 120, 5), "Sequence must advance");
Check(!state.Accept(1, 121, "猫", 8, out _), "Late result from old recording is rejected");
Check(!state.Accept(2, 125, "猫", 8, out _), "Expired recognition cannot place a stone");
state.Cancel();
Check(!state.Accept(2, 122, "猫", 8, out _), "Meeting or death cancellation blocks late results");
Check(state.Arm(3, 130, 5) && state.Accept(3, 135.4, "猫", 8, out _, 1), "In-time RPC has bounded delivery grace");
Check(state.Arm(4, 140, 5) && !state.Accept(4, 146, "猫", 8, out _, 1), "Delivery grace has a strict deadline");
state.Cancel();
Check(state.Arm(5, 160, 5), "Next recording");
state.Cancel();
Check(!state.Accept(5, 162, "猫", 8, out _) && !state.Waiting, "Cancellation ends pending recognition");
Check(!state.Arm(6, 170, double.NaN), "Reject invalid timer");
var stable = new VoiceStoneRequest();
Check(stable.Arm(1, 50.123456, 5, 1700000000000), "Stable deadline fixture");
Check(stable.SpeechExpiresUtc == 1700000005000, "UTC deadline captured once at arm, without polling clock conversion");
Check(stable.Accept(1, 51, "猫", 8, out _) && stable.Arm(2, 52, 3, 1700000002000), "Next recording has a stable deadline");
Check(stable.SpeechExpiresUtc == 1700000005000 && stable.Expires == 55, "New request snapshots both clocks once");
stable.Cancel();
Check(stable.SpeechExpiresUtc == 0, "Cancellation clears published deadline");
var ears = new WallEarRound();
Check(ears.Place(1, .9f, 0, 0, 3, 0) && ears.Place(2, .4f, 0, 0, 3, 0), "Ear fixtures");
Check(ears.Nearest(0, 0, _ => true)?.Id == 2, "Use button selects nearest ear");
Check(ears.Nearest(0, 0, e => e.Id != 2)?.Id == 1, "Cannot use an ear behind a wall");
Check(ears.Nearest(3, 0, _ => true) == null && ears.Nearest(float.NaN, 0, _ => true) == null, "No button target out of reach");
var sound = VoiceStoneImpactSynth.Create();
Check(sound.Length == 7056 && sound.All(float.IsFinite) && sound.Max(Math.Abs) <= .8f, "Impact length, finite samples, no clipping");
Check(sound[0] == 0 && sound[^1] == 0 && sound.Take(2000).Sum(v => v * v) > sound.TakeLast(1000).Sum(v => v * v) * 100, "Impact has an attack and fades out");
Console.WriteLine("PASS voice stone immediate placement, expiry/replay/cancellation, nearby ear selection and stone impact audio");

var neural = new VoiceStoneRequest();
Check(neural.Arm(1, 100, 5, 1700000000000, true) && neural.LocalNeural && neural.Expires == 125, "Shared local model has a bounded recording and response window");
Check(neural.RecordUntilUtc == 1700000005000 && neural.SpeechExpiresUtc == 1700000025000, "Record window and recognition deadline stay separate");
Check(neural.Accept(1, 102, "インポスター", 8, out _), "Shared-model result places before the recording limit");
Check(neural.Arm(2, 111, 3, 1700000011000) && !neural.LocalNeural && neural.Expires == 114, "Next recording can use the lightweight fallback");
neural.Cancel(); Check(!neural.LocalNeural && !neural.Accept(2,112,"猫",8,out _), "Cancellation blocks stale local-model results");
Check(neural.Arm(3, 120, 5, 1700000020000, true) && neural.Accept(3,130,"猫",8,out _), "Shared-model result may arrive after the maximum recording duration");
