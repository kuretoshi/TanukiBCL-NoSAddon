using TanukiBCL.Roles;

void Check(bool ok, string name) { if (!ok) throw new Exception(name); }
Check(VoiceStoneRules.TryWord("ねこ", 8, out var word) && word == "ねこ", "Japanese word");
Check(VoiceStoneRules.Width("インポスター", .5f) == 3f && VoiceStoneRules.Width("ねこ", .5f) == 1f, "Character-proportional width");
Check(VoiceStoneRules.TryWord("ＴＡＳＫ", 4, out word) && word == "TASK", "Compatibility normalization");
foreach (var invalid in new[] { "", "猫 犬", "<b>猫", "猫\n", "🐈", "長すぎる単語です", "\ud800" })
    Check(!VoiceStoneRules.TryWord(invalid, 4, out _), "Reject invalid or oversized word: " + invalid);
var state = new VoiceStoneRequest();
Check(state.Arm(1, 100, 5), "Begin listening");
Check(!state.Arm(2, 101, 5), "Cannot replace an active request");
Check(!state.Accept(2, 102, "猫", 8, out _), "Reject another request");
Check(!state.Accept(1, 102, "猫", 8, out _) && state.Waiting, "Speech alone cannot place a stone");
Check(!state.Propose(1, 102, "猫 犬", 8) && state.Waiting, "Bad word does not consume request");
Check(state.Propose(1, 103, "ポスター", 8) && state.Confirming && !state.Waiting, "Recognition only opens confirmation");
Check(!state.Accept(1, 104, "インポスター", 8, out _), "Cannot substitute an unrecognised word at placement");
Check(state.Retry(1, 2, 104, 5) && state.Waiting && !state.Confirming, "Re-record without accepting wrong text");
Check(!state.Propose(1, 105, "ポスター", 8), "Late result from old recording is rejected");
Check(state.Propose(2, 105, "猫", 8), "New recording produces a new candidate");
Check(!state.Arm(3, 106, 5), "Cannot bypass confirmation by arming again");
Check(state.Accept(2, 112, "猫", 8, out word) && word == "猫", "Confirm after original recording timer ends");
Check(!state.Accept(2, 113, "猫", 8, out _), "Replay cannot create a second stone");
Check(!state.Arm(2, 120, 5) && state.Arm(3, 120, 5), "Sequence must advance");
Check(!state.Propose(3, 125, "猫", 8), "Expired speech cannot create a candidate");
state.Cancel();
Check(!state.Propose(3, 122, "猫", 8), "Meeting or death cancellation blocks late results");
Check(state.Arm(4, 130, 5) && state.Propose(4, 135.4, "猫", 8, 1), "In-time RPC has bounded delivery grace");
Check(!state.Accept(4, 156.4, "猫", 8, out _, 1), "Confirmation expires at its bounded deadline");
state.Cancel();
Check(state.Arm(5, 160, 5) && state.Propose(5, 161, "猫", 8), "Next confirmation");
state.Cancel();
Check(!state.Accept(5, 162, "猫", 8, out _) && !state.Confirming, "Cancellation closes pending confirmation");
Check(!state.Arm(6, 170, double.NaN), "Reject invalid timer");
var stable = new VoiceStoneRequest();
Check(stable.Arm(1, 50.123456, 5, 1700000000000), "Stable deadline fixture");
Check(stable.SpeechExpiresUtc == 1700000005000, "UTC deadline captured once at arm, without polling clock conversion");
Check(stable.Propose(1, 51, "猫", 8) && stable.Retry(1, 2, 52, 3, 1700000002000), "Stable retry deadline");
Check(stable.SpeechExpiresUtc == 1700000005000 && stable.Expires == 55, "Retry snapshots both clocks once");
stable.Cancel();
Check(stable.SpeechExpiresUtc == 0, "Cancellation clears published deadline");
var ears = new WallEarRound();
Check(ears.Place(1, .9f, 0, 0, 3, 0) && ears.Place(2, .4f, 0, 0, 3, 0), "Ear fixtures");
Check(ears.Nearest(0, 0, _ => true)?.Id == 2, "Use button selects nearest ear");
Check(ears.Nearest(0, 0, e => e.Id != 2)?.Id == 1, "Cannot use an ear behind a wall");
Check(ears.Nearest(3, 0, _ => true) == null && ears.Nearest(float.NaN, 0, _ => true) == null, "No button target out of reach");
var sound = VoiceStoneImpactSynth.Create();
Check(sound.Length == 7056 && sound.All(float.IsFinite) && sound.Max(Math.Abs) <= .8f, "Impact length, finite samples, no clipping");
Check(sound[0] == 0 && sound[^1] == 0 && sound.Take(2000).Sum(v => v * v) > sound.TakeLast(1000).Sum(v => v * v) * 100, "Impact has an attack and fades out");
Console.WriteLine("PASS voice stone confirmation/re-record, expiry/replay/cancellation, nearby ear selection and stone impact audio");
