using TanukiBCL.Roles;
using Nebula.Modules.Cosmetics;
using UnityEngine;
void Check(bool valid, string message) { if (!valid) throw new Exception(message); }
var progress = new BitCharacterProgress(5,10,15,20,25);
Check(progress.Stage == 0 && progress.Remaining == 5, "starts in 2bit");
Check(!progress.Advance(60,false) && progress.Elapsed == 0, "meeting, death and intro pause");
Check(!progress.Advance(4.99,true), "before exact boundary");
Check(progress.Advance(.01,true) && progress.Stage == 1 && progress.Remaining == 10, "exact 4bit boundary");
Check(progress.Advance(25,true) && progress.Stage == 3 && progress.Remaining == 20, "cross multiple stages");
Check(!progress.Advance(300,false) && progress.Stage == 3, "meeting preserves partial timeline");
Check(!progress.Advance(double.NaN,true) && !progress.Advance(-1,true), "invalid delta rejected");
Check(progress.Advance(44,true) && progress.Stage == 4 && progress.Remaining == 1 && BitCharacterProgress.Bits[4] == 32, "32bit is the final active stage at total74s");
Check(!progress.Advance(300,false) && progress.Remaining == 1, "meeting pauses the 32bit stage");
Check(progress.Advance(1,true) && progress.Stage == BitCharacterProgress.FinalStage && progress.Remaining == 0, "32bit restores original directly at total75s without 64bit");
Check(BitCharacterProgress.FinalStage==6 && !BitCharacterProgress.IsValidStage(5),"retired 64bit slot is skipped while the published normal-voice wire value stays compatible");
Check(!progress.Advance(100,true), "final stage stable");
try { new BitCharacterProgress(0,1,2,3,4); throw new Exception("accepted zero duration"); } catch(ArgumentException) { }
try { new BitCharacterProgress(1,2,3,4,5,6); throw new Exception("accepted removed 64bit duration"); } catch(ArgumentException) { }
AddonLinkState.Reset(); BitCharacterBroadcast.Set(1,0); BitCharacterBroadcast.Set(2,4);
var rows=AddonLinkState.Snapshot(DateTimeOffset.UtcNow.ToUnixTimeMilliseconds())["bits"];
Check(rows.GetArrayLength()==2 && rows[1].GetProperty("stage").GetInt32()==4,"multiple modifier holders share link");
BitCharacterBroadcast.Remove(1);
Check(AddonLinkState.Snapshot(DateTimeOffset.UtcNow.ToUnixTimeMilliseconds())["bits"].GetArrayLength()==1,"removal preserves others");
BitCharacterBroadcast.Remove(2);
Check(AddonLinkState.Snapshot(DateTimeOffset.UtcNow.ToUnixTimeMilliseconds())["bits"].GetArrayLength()==0,"last removal clears");
BitCharacterBroadcast.Set(3,5);
Check(AddonLinkState.Snapshot(DateTimeOffset.UtcNow.ToUnixTimeMilliseconds())["bits"].GetArrayLength()==0,"retired 64bit cannot be broadcast");
BitCharacterBroadcast.Set(3,6);
Check(AddonLinkState.Snapshot(DateTimeOffset.UtcNow.ToUnixTimeMilliseconds())["bits"][0].GetProperty("stage").GetInt32()==6,"final original stage broadcast");
BitCharacterBroadcast.Remove(3);
foreach(string name in new[]{"BitCharacter2","BitCharacter4","BitCharacter32","SuperHero","EvilHero"}) MoreCosmic.AllHats[CosmicHat.IdToProductId("TanukiBCL_"+name)]=new CosmicHat(name);
var player=new PlayerControl();
var bit=new HeroCostume(player,"BitCharacter2",true);bit.Update();
Check(player.cosmetics.hat.Hat!.ProductId==CosmeticsLayer.EMPTY_HAT_ID && player.Body.forceRenderingOff,"bit applies and hides normal body");
bit.LateUpdate();Check(player.cosmetics.hat.SpriteSyncNode!.enabled,"native walk synchronizer preserved");
var hero=new HeroCostume(player,HeroKind.Super);hero.Update();
Check(player.cosmetics.hat.Hat!.ProductId.EndsWith("SuperHero"),"hero takes priority");
bit.Dispose();
var nextBit=new HeroCostume(player,"BitCharacter4",true);nextBit.Update();nextBit.LateUpdate();
Check(player.cosmetics.hat.Hat!.ProductId.EndsWith("SuperHero"),"stage change cannot overwrite hero");
hero.Dispose();Check(player.cosmetics.hat.Hat!.ProductId==CosmeticsLayer.EMPTY_HAT_ID,"hero expiry resumes current bit stage");
nextBit.Dispose();
Check(player.cosmetics.hat.Hat!.ProductId=="original-hat" && player.cosmetics.Skin=="original-skin" && player.cosmetics.Visor=="original-visor","last costume restores actual wardrobe");
Check(!player.Body.forceRenderingOff && player.cosmetics.hat.SpriteSyncNode!.enabled,"original rendering and synchronizer restored");
player.Body.forceRenderingOff=true;
using(var b=new HeroCostume(player,"BitCharacter2",true)) b.Update();
Check(player.Body.forceRenderingOff,"preexisting hidden state preserved");
player.Body.forceRenderingOff=false;
using(var smooth=new HeroCostume(player,"BitCharacter32",true))
{
    Check(player.cosmetics.hat.Hat!.ProductId==CosmeticsLayer.EMPTY_HAT_ID,"32bit uses native body without an overlaid full-body hat");
    using(var evil=new HeroCostume(player,HeroKind.Evil))
    {
        smooth.Update();
        Check(player.cosmetics.hat.Hat!.ProductId.EndsWith("EvilHero"),"32bit still yields to a hero costume");
    }
    Check(player.cosmetics.hat.Hat!.ProductId==CosmeticsLayer.EMPTY_HAT_ID,"hero expiry resumes native 32bit");
}
Check(player.cosmetics.hat.Hat!.ProductId=="original-hat" && !player.Body.forceRenderingOff,"32bit expiry restores original wardrobe/body");
foreach(string name in new[]{"BitCharacter8","BitCharacter16","BitCharacter32"}) MoreCosmic.AllHats[CosmicHat.IdToProductId("TanukiBCL_"+name)]=new CosmicHat(name);
int[] held={0,0,0,0,4,4,6,6,6,6,10,10};
for(int i=0;i<12;i++)
{
    foreach(int bits in new[]{2,4,8}) Check(BitCharacterMotion.WalkFrame(i/12f,bits)==held[i],"native stride/contact order retained");
    foreach(int bits in new[]{16,32}) Check(BitCharacterMotion.WalkFrame(i/12f,bits)==i,"higher stage full native walk");
}
Check(BitCharacterMotion.WalkTick(1.1f)>BitCharacterMotion.WalkTick(.1f),"new walk cycles cannot reuse a stale held pose");
Check(BitCharacterMotion.WalkTick(float.NaN)==long.MinValue && BitCharacterMotion.WalkTick(float.PositiveInfinity)==long.MinValue,"invalid animation phase releases holding");
Console.WriteLine("PASS bit stages: exact boundaries, pauses, direct 32bit-to-normal transition, legacy voice compatibility, multiple holders and costume priority/lifetime/wardrobe restoration");
NativeVisualTests.Run();
