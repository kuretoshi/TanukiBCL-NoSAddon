static async Task Write(NetworkStream stream, object value) => await stream.WriteAsync(Encoding.UTF8.GetBytes(JsonSerializer.Serialize(value) + "\n"));
static async Task Fixture()
{
    using var server = new AddonLinkServer(Environment.ProcessId);
    Console.WriteLine(JsonSerializer.Serialize(new { pid = Environment.ProcessId, port = server.Port, session = server.Session }));
    Console.Out.Flush();
    var input = Task.Run(() => Console.In.ReadLine()); long revision = 0;
    string phase = "tasks"; int local = 1; bool host = true; JsonElement? custom = null;
    while (true)
    {
        if (input.IsCompleted)
        {
            string? line = await input; if (line == null) break;
            using var doc = JsonDocument.Parse(line); var m = doc.RootElement;
            if (m.TryGetProperty("channels", out var c)) custom = c.Clone();
            if (m.TryGetProperty("phase", out var p)) phase = p.GetString()!;
            if (m.TryGetProperty("localPlayer", out var l)) local = l.GetInt32();
            if (m.TryGetProperty("host", out var h)) host = h.GetBoolean();
            input = Task.Run(() => Console.In.ReadLine());
        }
        var channels = custom ?? Json(JsonSerializer.Serialize(AddonLinkState.Snapshot(Now())));
        server.SetSnapshot(JsonSerializer.Serialize(new { kind = "state", protocol = 1, pid = Environment.ProcessId,
            session = server.Session, sequence = ++revision, unixMs = Now(), phase, localPlayer = local, host, channels }), revision);
        while (server.TryRead(out var command)) Console.WriteLine(JsonSerializer.Serialize(new { command }));
        Console.Out.Flush(); await Task.Delay(50);
    }
}
